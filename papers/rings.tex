\section{Formal Model}

\begin{assumption}[System scope]
Rings is modeled as a fail-stop structured overlay with authenticated participants, signed protocol messages, asynchronous delivery, and runtime-specific IO interpreters. Byzantine consensus is not a base-overlay operation. Stronger ordering is represented by sidecar witnesses. Cryptographic multiplication, threshold interpolation, pairings, and subgroup checks live in explicit cryptographic carriers, not in the Chord identifier carrier.
\end{assumption}

\begin{definition}[Identifier carrier]
Let
\[
  M=2^{160},\qquad R=\mathbb{Z}/M\mathbb{Z}.
\]
For $a,x\in R$, define clockwise distance
\[
  \delta_a(x)=(x-a)\bmod M
\]
and circular interval
\[
  (a,b]_R=\{x\in R:0<\delta_a(x)\leq\delta_a(b)\}.
\]
The overlay uses $(R,+,0,-)$ as an additive cyclic group. Multiplication in $R$ is outside the routing model.
\end{definition}

\begin{definition}[Live topology]
For a nonempty live set $N\subset R$ and $n,k\in R$,
\[
  \operatorname{owner}_N(k)
    =\arg\min_{u\in N}\delta_k(u),
\]
\[
  \operatorname{pred}_N(n)
    =\arg\min_{p\in N\setminus\{n\}}\delta_p(n),
\]
\[
  \operatorname{Succ}^q_N(n)
    =
    \operatorname{take}_q
    \bigl(\operatorname{sort}_{\delta_n}(N\setminus\{n\})\bigr).
\]
Argmins are unique because $\delta_a$ is injective on $N$.
\end{definition}

\begin{assumption}[Correct-Chord stable base]
Let $r$ be the successor-list length. Steady-state algebra assumes
\[
  |N|\geq r+1.
\]
Bootstrap states with fewer members are initialization states. The maintenance reference is Zave's Correct Chord transition family of join, rectify, and stabilize \cite{ZaveChordCorrect}.
\end{assumption}

\begin{definition}[Replica set]
For replication factor $r$,
\[
  \operatorname{Rep}^r_N(k)=
  \{\operatorname{owner}_N(k)\}
  \cup
  \mathrm{set}\bigl(
    \operatorname{Succ}^{r-1}_N(\operatorname{owner}_N(k))
  \bigr).
\]
Thus the owner is included at index $0$ of storage responsibility.
\end{definition}

\begin{definition}[Rings overlay object]
A steady Rings overlay state over $N$ is
\[
  \mathcal{R}_N=
  (R,N,\operatorname{owner}_N,\operatorname{pred}_N,
   \operatorname{Succ}^r_N,\operatorname{Rep}^r_N,F,E),
\]
where $F$ is the sparse finger relation and $E$ is the replicated entry algebra. The safety root is $(\operatorname{pred}_N,\operatorname{Succ}^r_N)$; fingers are routing witnesses.
\end{definition}

\begin{proposition}[Rotation equivariance]
For $t\in R$, $N+t=\{n+t:n\in N\}$ satisfies
\[
  \operatorname{owner}_{N+t}(k+t)=\operatorname{owner}_N(k)+t.
\]
\end{proposition}
\begin{proof}
For each $n\in N$,
\[
  \delta_{k+t}(n+t)=((n+t)-(k+t))\bmod M=\delta_k(n).
\]
Translation preserves the order used by $\arg\min$.
\end{proof}

\begin{constraint}[Carrier separation]
Overlay placement may use only
\[
  0,\quad +,\quad -,\quad \delta_a,\quad (a,b]_R,\quad
  \operatorname{owner}_N,\quad \operatorname{Rep}^r_N.
\]
A protocol requiring multiplication, inversion, scalar multiplication, polynomial interpolation, pairings, or subgroup checks names a separate carrier such as a finite field $\mathbb{F}_q$ or elliptic-curve group $\mathbb{G}$.
\end{constraint}

\begin{definition}[Identity proof]
A DID proof family is a tuple
\[
  \Pi=(K,\Sigma,V,D,\mathsf{enc},\mathsf{ttl})
\]
where $K$ is the public-key space, $\Sigma$ is the signature space, $V:K\times\Sigma\times B^*\to\{\top,\bot\}$ is verification, $D:K\to R$ is DID derivation, $\mathsf{enc}$ is signature encoding, and $\mathsf{ttl}$ bounds replay.
\end{definition}

\begin{invariant}[Session authority]
For account DID $d$ and delegated session key $s$,
\[
  \operatorname{Session}(d,s,e)
  \Rightarrow
  V(k_d,\sigma_{d\to s},d\Vert s\Vert e)=\top
  \land now<e.
\]
Every hop may check session authority without resolving a W3C DID document in the overlay path.
\end{invariant}

\begin{definition}[Transport interface]
Transport is modeled as
\[
  T=(C,\mathsf{offer},\mathsf{answer},\mathsf{send},\mathsf{recv})
\]
with channel state $C$ and action alphabet
\[
  A_T=\{\mathsf{Offer},\mathsf{Answer},\mathsf{Ice},\mathsf{Open},
        \mathsf{Send},\mathsf{Recv},\mathsf{Close}\}.
\]
WebRTC and native transports are interpreters of $A_T^*$.
\end{definition}

\begin{obligation}[Runtime confinement]
For every extension transition $\tau$ and interpreter $I_X$,
\[
  \tau:S\times I\to \mathcal{E}+(S\times O\times A_X^*)
\]
may emit only actions in $\mathsf{cap}$. The interpreter rejects $a\notin\mathsf{cap}$ before IO.
\end{obligation}

\section{Category and Effects}

\begin{definition}[Namespace object]
A namespace $\nu$ is an object of the slice category $\mathbf{Set}/R$:
\[
  H_\nu:X_\nu\to R.
\]
DIDs, VIDs, mailbox roots, service descriptors, subring anchors, and ranking keys are distinct namespace objects over the same carrier.
\end{definition}

\begin{definition}[Placement functor]
For live topology $N$, define
\[
  \Omega_N:\mathbf{Set}/R\to\mathbf{Set}/N
\]
by
\[
  \Omega_N(X,H)=(X,\operatorname{owner}_N\circ H).
\]
The placement map of namespace $\nu$ is
\[
  P^N_\nu=\operatorname{owner}_N\circ H_\nu:X_\nu\to N.
\]
\end{definition}

\begin{proposition}[Placement functoriality]
If $f:X_\nu\to X_\mu$ satisfies $H_\mu\circ f=H_\nu$, then
\[
  P^N_\mu\circ f=P^N_\nu.
\]
\end{proposition}
\begin{proof}
Post-compose $H_\mu\circ f=H_\nu$ with $\operatorname{owner}_N$.
\end{proof}

\begin{definition}[Writer effect monad]
For action alphabet $A$, let $A^*$ be the free monoid of finite action lists. Define
\[
  \mathsf{T}_A(X)=X\times A^*,
\]
\[
  \eta(x)=(x,\epsilon),
\]
\[
  (x,\alpha)\mathbin{\gg\!=}f
    =
  \text{let } f(x)=(y,\beta)
  \text{ in }(y,\alpha\cdot\beta).
\]
\end{definition}

\begin{proposition}[Effect associativity]
Kleisli composition for $\mathsf{T}_A$ is associative.
\end{proposition}
\begin{proof}
The accumulated effect is $\alpha\cdot\beta\cdot\gamma$; associativity follows from associativity of the free monoid $A^*$.
\end{proof}

\begin{definition}[Typed failure stack]
Rings uses strict error-over-writer composition:
\[
  \mathsf{T}^{\mathcal{E}}_A(X)=\mathcal{E}+(X\times A^*).
\]
An error branch has no executable action list. Recoverable reports are ordinary successful actions.
\end{definition}

\begin{definition}[Protocol object]
A protocol namespace is
\[
  \mathcal{P}_\nu=
  (X_\nu,S_\nu,I_\nu,O_\nu,A_\nu,\mathcal{E}_\nu,H_\nu,\tau_\nu)
\]
with transition
\[
  \tau_\nu:S_\nu\times I_\nu
  \to
  \mathsf{T}^{\mathcal{E}_\nu}_{A_\nu}(S_\nu\times O_\nu).
\]
\end{definition}

\begin{definition}[Protocol morphism]
A morphism $\Sigma:\mathcal{P}_\nu\to\mathcal{P}_\mu$ is a family
\[
  (\sigma_X,\sigma_S,\sigma_I,\sigma_O,\sigma_{\mathcal{E}},\sigma_A^*)
\]
where $\sigma_A^*:A_\nu^*\to A_\mu^*$ is a monoid homomorphism and
\[
  H_\mu\circ\sigma_X=H_\nu,
\]
\[
  \mathsf{T}_{\sigma_A}^{\sigma_{\mathcal{E}}}
    (\sigma_S\times\sigma_O)
  \circ\tau_\nu
  =
  \tau_\mu\circ(\sigma_S\times\sigma_I).
\]
\end{definition}

\begin{definition}[Local node state]
For node $n\in N$,
\[
  S_n=(p_n,L_n,F_n,E_n)
\]
where $p_n$ is predecessor, $L_n$ is successor list, $F_n$ is sparse finger table, and $E_n$ is local entry evidence. The fixpoint target is
\[
  p_n=\operatorname{pred}_N(n),
  \qquad
  L_n=\operatorname{Succ}^r_N(n).
\]
\end{definition}

\begin{definition}[Finger target]
For slot $i$,
\[
  \operatorname{target}_i(n)=n+2^i,
\]
\[
  \operatorname{finger}_i(n)=
  \operatorname{owner}_N(\operatorname{target}_i(n)).
\]
Sparse no-wrap state permits $F_n(i)=\bot$; present fingers are performance witnesses.
\end{definition}

\begin{algorithm}
\caption{DHT Join Transitions}
\label{alg:join}
\begin{algorithmic}[1]
\Function{BeginJoin}{$n, known$}
\State $pred[n]\gets none$
\State $a\gets FindSuccessor(n.did)$
\State \Return $Remote(known, a)$
\EndFunction
\Function{InstallSuccessor}{$n, s$}
\State $succ[n]\gets take(r, [s])$
\State \Return $Remote(s, QuerySuccList(s))$
\EndFunction
\Function{InstallSuccList}{$n, s, list$}
\State $succ[n]\gets take(r, s :: list)$
\If{$head(succ[n]) \neq none$}
\State $h\gets head(succ[n])$
\State $notify\gets Notify(h, n.did)$
\State $sync\gets SyncStorage(h)$
\State \Return $Multi(notify, sync)$
\Else
\State \Return $Noop$
\EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{DHT Lookup Algorithm}
\label{alg:lookup}
\begin{algorithmic}[1]
\Function{Lookup}{$key, node$}
    \State $s\gets head(succ[node])$
    \If{$s = none$}
        \State \Return $Repair(QuerySuccList(node))$
    \EndIf
    \If{$key \in (node.did, s]$}
        \State \Return $Local(s)$
    \EndIf
    \State $next\gets closest\_preceding\_node(node, key)$
    \If{$next = none$}
        \State \Return $Repair(QuerySuccList(s))$
    \Else
        \State $a\gets FindSuccessor(key)$
        \State \Return $Remote(next, a)$
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{DHT Stabilization}
\label{alg:stabilize}
\begin{algorithmic}[1]
\Procedure{Stabilize}{$n, topo$}
\State $actions\gets []$
\State $c\gets topo.pred$
\State $s\gets head(succ[n])$
\If {$c \neq none$ and $c \in (n.did, s)$}
\State $succ[n]\gets insert(succ[n], c)$
\State $actions\gets actions :: FindSuccessor(c)$
\EndIf
\State $succ[n]\gets take(r, merge(succ[n], topo.succ))$
\If{$head(succ[n]) \neq none$}
\State $h\gets head(succ[n])$
\State $actions\gets actions :: Notify(h, n.did)$
\EndIf
\State \Return $Multi(actions)$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{invariant}[Ring fixpoint]
\[
\begin{aligned}
  \operatorname{Inv}_{ring}(N,S)
  \equiv \forall n\in N.\;&
  p_n=\operatorname{pred}_N(n)\\
  &\land L_n=\operatorname{Succ}^r_N(n).
\end{aligned}
\]
\end{invariant}

\begin{invariant}[Lookup owner preservation]
For steady topology $N$,
\[
  \operatorname{Lookup}(S,k)\Downarrow
  \operatorname{owner}_N(k).
\]
The judgment $\Downarrow$ is reachability in the action-interpreter transition relation generated by Algorithms \ref{alg:lookup} and \ref{alg:stabilize}.
\end{invariant}

\section{DID and Resource Algebra}

\begin{definition}[DID namespace]
A DID namespace is $H_D:X_D\to R$. Account-controlled DIDs add proof family $\Pi$ and session relation $\operatorname{Session}(d,s,e)$.
\end{definition}

\begin{definition}[VID namespace]
A VID namespace is $H_V:X_V\to R$ with no private key attached to $H_V(x)$. Authorization is carried by entry operations:
\[
  \operatorname{Auth}_V(op,x,\sigma,caps)\in\{\top,\bot\}.
\]
\end{definition}

\begin{definition}[Biased order]
For fixed point $C$,
\[
  A\leq_C B
  \Longleftrightarrow
  \delta_C(A)\leq\delta_C(B).
\]
\end{definition}

\begin{algorithm}[htbp]
\caption{Biased Circular Comparison}
\label{alg:biasdid}
\begin{algorithmic}[1]
\Function{BiasedCompare}{$A,B,C$}
\State $a\gets \delta_C(A)$
\State $b\gets \delta_C(B)$
\If{$a < b$}
\State \Return $A <_C B$
\ElsIf{$a > b$}
\State \Return $B <_C A$
\Else
\State \Return $A = B$
\EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{definition}[Replicated entry algebra]
For each VID $v$, entry state is either a join semilattice
\[
  (E_v,\sqsubseteq,\sqcup,\bot)
\]
with commutativity, associativity, idempotence, and bottom identity, or a pair
\[
  (E_v,\mathsf{conflict}_v,\mathsf{finalize}_v)
\]
that resolves conflicts explicitly. The semilattice case follows CRDT merge discipline \cite{ShapiroCRDT}.
\end{definition}

\begin{invariant}[Placement]
\[
  \operatorname{Inv}_{place}\equiv
  \forall \nu,x.\;
  \operatorname{replicas}(H_\nu(x))
  \subseteq
  \operatorname{Rep}^r_N(H_\nu(x)).
\]
\end{invariant}

\begin{definition}[Mailbox]
For DID $d$ and epoch $e$,
\[
  mbox(d,e)=H_M(\mathtt{rings.mailbox}\Vert d\Vert e).
\]
Mailbox entries are encrypted envelopes with nonce, expiration, sender proof, recipient DID, and ciphertext.
\end{definition}

\begin{definition}[Hidden service descriptor]
A service descriptor is
\[
  h=(\nu,\rho,Intro,Relay,Cap,K,exp,rank).
\]
Its VID is $H_H(h)$; lookup returns placement, while authorization is $\operatorname{Auth}_H(h,\sigma)$.
\end{definition}

\begin{definition}[Subring]
A subring is
\[
  \mathcal{S}_u=(u,N_u,\operatorname{owner}_{N_u},
  \operatorname{Succ}^{r_u}_{N_u},E_u)
\]
where $u=H_U(policy)$ anchors membership. Membership operations form a mergeable log or finalized state.
\end{definition}

\section{Cryptographic Carriers and Traffic}

\begin{definition}[End-to-end carrier]
Let $\mathbb{G}$ be an elliptic-curve group of order $q$ with scalar field $\mathbb{Z}_q$. Private key $x\in\mathbb{Z}_q$ has public key $h=g^x$. This carrier is distinct from $R$.
\end{definition}

\begin{algorithm}[h]
\caption{ElGamal Encryption and Decryption}
\label{alg:elgamal}
\begin{algorithmic}[1]
\State \textbf{Input:} Message $m$, private key $x \in \mathbb{Z}_q$, public key $h = g^x$, generator $g$
\State \textbf{Output:} Encrypted message $(c_1, c_2)$
\Procedure{Encryption}{}
\State Choose a random integer $k \in \mathbb{Z}_q$
\State Compute $c_1 = g^k$
\State Compute $c_2 = m\cdot h^k$
\State \Return $(c_1, c_2)$
\EndProcedure
\Procedure{Decryption}{}
\State Compute $m = c_2\cdot(c_1)^{-x}$
\State \Return $m$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{definition}[Signed encrypted frame]
An encrypted frame is
\[
  f=(sid,seq,last,cipher,pk_s,\sigma_s).
\]
Validity is
\[
\begin{aligned}
  &V(pk_s,\sigma_s,
    sid\Vert seq\Vert last\Vert cipher)=\top\\
  &\land\;
    \operatorname{Session}(D(pk_s),sid,e).
\end{aligned}
\]
\end{definition}

\begin{definition}[Secret sharing placement]
For threshold polynomial $P\in\mathbb{F}_q[X]$, shares are
\[
  sh_i=(i,P(i))\in\mathbb{F}_q^2.
\]
Overlay placement is $H_S(sh_i)\in R$; reconstruction is field interpolation in $\mathbb{F}_q$ \cite{ssss}.
\end{definition}

\begin{definition}[Relay frame]
A hop-by-hop frame is
\[
  r=(path,dst,\nu,session,payload,report).
\]
Rings borrows message splitting vocabulary from MSRP \cite{RFC4975}, while relay validity is native to this tuple.
\end{definition}

\begin{algorithm}[h]
\caption{End-to-End Chunking Algorithm}
\label{alg:e2e_chunking}
\begin{algorithmic}[1]
\Procedure{E2E Chunking}{}
\State Input: data, chunk size
\State Output: chunks
\State chunks $\gets$ []
\State total-chunks $\gets$ ceil(len(data)/chunk-size)
\For{i in range(total-chunks)}
\State chunk $\gets$ data[i*chunk-size : (i+1)*chunk-size]
\State append chunk to chunks
\EndFor
\State \Return chunks
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{invariant}[Chunk reassembly]
For message $m$ with chunks $c_i$,
\[
\begin{aligned}
  \operatorname{complete}(m)
  \Leftrightarrow\;&
  \forall i<total(m).\;\exists!c_i\\
  &\land \sum_i |c_i|\leq B_m\\
  &\land now<ttl(m).
\end{aligned}
\]
\end{invariant}

\section{Ordering, Ranking, and Security}

\begin{definition}[Sidecar witness]
A witness is
\[
  w=(domain,root,height,time,finality).
\]
For message commitment $c=H(msg)$,
\[
\begin{aligned}
  \operatorname{Witness}(msg,w)
  \Leftrightarrow\;&
  \operatorname{Include}(c,w.root)=\top\\
  &\land \operatorname{Final}(w)=\top.
\end{aligned}
\]
\end{definition}

\begin{definition}[Witness order]
For messages $a,b$ with witnesses $w_a,w_b$,
\[
\begin{aligned}
  a\prec b
  \Leftrightarrow\;&
  domain(w_a)=domain(w_b)\\
  &\land time(w_a)<time(w_b).
\end{aligned}
\]
Different domains require an application comparison morphism
\[
  \chi:Domain_a\times Domain_b\to Order.
\]
\end{definition}

\begin{definition}[Ranking event]
A ranking event is
\[
\begin{aligned}
  q=(&subject,rater,behavior,\\
     &evidence,epoch,weight,\sigma).
\end{aligned}
\]
Its validity predicate is
\[
\begin{aligned}
  m_q &= subject\Vert behavior\Vert evidence\Vert epoch,\\
  \operatorname{RankValid}(q)
    &\Leftrightarrow V(k_{rater},\sigma,m_q)=\top.
\end{aligned}
\]
\end{definition}

\begin{definition}[Score aggregation]
For epoch $e$ and subject $s$,
\[
\begin{aligned}
  score_e(s)=
  \operatorname{robust}\{&
  weight(q)\cdot value(q):\\
  &q.subject=s\}.
\end{aligned}
\]
The aggregation function is median, trimmed mean, or another declared robust statistic.
\end{definition}

\begin{obligation}[Replay rejection]
For session $s$ and message nonce $\eta$,
\[

\end{obligation}

\section{Verification and Evaluation}

\begin{definition}[State relation]
A model configuration is
\[
  C=(S,Q)
\]
where $S$ is protocol state and $Q\in A^*$ is the pending action list. The step relation
\[
  C\to C'
\]
interprets the head of $Q$, applies a pure transition $\tau$, and appends emitted actions. This is the TLA-style state relation used for finite model exploration \cite{LamportSpecifyingSystems,Stateright}.
\end{definition}

\begin{invariant}[Effect refinement]
For implementation path $\operatorname{Impl}_A$ and action interpreter $I_A^\sharp$,
\[
  \operatorname{Impl}_A(s,i)
  \preceq
  I_A^\sharp(\tau(s,i)).
\]
Runtime scheduling and logging are refinement details; protocol state changes factor through $\tau$.
\end{invariant}

\begin{obligation}[Topology coverage]
The model-checking domain contains representative layouts
\[
  \mathcal{L}=\{line,wrap,gap,partition,merge,churn\}
\]
and schedule classes
\[
  \mathcal{S}=\{loss,dup,reorder,timeout,restart\}.
\]
Each checked property names $(L,S,P)$ with $L\in\mathcal{L}$, $S\in\mathcal{S}$, and predicate $P$.
\end{obligation}

\begin{definition}[Latency decomposition]
End-to-end latency is modeled as
\[
  T_{total}
  =
  T_{conn}
  +h\cdot T_{hop}
  +T_{crypto}
  +T_{app}
  +T_{witness}.
\]
For Chord lookup, $h=O(\log |N|)$ under populated fingers.
\end{definition}

\section{Related Work}

Kademlia-derived networks use XOR distance rather than circular successor order \cite{Kademlia,IPFS,Libp2p}. Tor and Nym emphasize relay anonymity rather than structured ownership \cite{TorNetwork,NymNetwork}. Rings selects Chord because the functions $\operatorname{owner}_N$, $\operatorname{Succ}^r_N$, and $\operatorname{Rep}^r_N$ give DID and VID placement a small algebraic core.

\section{Conclusion}

Rings is the tuple
\[
  \mathcal{Rings}=
  (R,\mathcal{R}_N,\mathbf{Set}/R,\Omega_N,
   \mathsf{T}^{\mathcal{E}}_A,\Pi,\mathbb{G},\mathcal{X}).
\]
Its base overlay is Correct Chord over an additive circular carrier. Its resources are namespace objects. Its protocol transitions are pure state morphisms with explicit action evidence. Its privacy protocols use separate cryptographic carriers. Its evaluation artifacts are tables, figures, algorithms, invariants, and obligations tied to the same model.
